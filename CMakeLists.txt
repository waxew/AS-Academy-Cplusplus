cmake_minimum_required(VERSION 3.20)
project(ASAcademyCpp VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(AS_ACADEMY_BUILD_EXAMPLES "Build lesson examples" ON)
option(AS_ACADEMY_BUILD_PROJECTS "Build practice projects" ON)
option(AS_ACADEMY_BUILD_TESTS "Build smoke tests" ON)

if(AS_ACADEMY_BUILD_EXAMPLES)
  add_executable(cpp_hello fundamentals/01_hello_world/main.cpp)
  add_executable(cpp_variables fundamentals/02_variables/main.cpp)
  add_executable(cpp_io fundamentals/03_input_output/main.cpp)
  add_executable(cpp_control beginner/01_control_flow/main.cpp)
  add_executable(cpp_functions beginner/02_functions/main.cpp)
  add_executable(cpp_arrays beginner/03_arrays_strings/main.cpp)
  add_executable(cpp_pointers beginner/04_pointers/main.cpp)
  add_executable(cpp_oop intermediate/01_oop/main.cpp)
  add_executable(cpp_stl intermediate/02_stl/main.cpp)
  add_executable(cpp_files intermediate/03_file_io/main.cpp)
  add_executable(cpp_exceptions intermediate/04_exceptions/main.cpp)
  add_executable(cpp_templates advanced/01_templates/main.cpp)
  add_executable(cpp_smart advanced/02_smart_pointers/main.cpp)
  add_executable(cpp_threads advanced/03_threads/main.cpp)
  add_executable(cpp_algorithms advanced/04_algorithms/main.cpp)
endif()

if(AS_ACADEMY_BUILD_PROJECTS)
  add_executable(project_calculator projects/01_calculator/main.cpp)
  add_executable(project_students projects/02_student_manager/main.cpp)
  add_executable(project_library projects/03_library_manager/main.cpp)
endif()

include(CTest)
if(BUILD_TESTING AND AS_ACADEMY_BUILD_TESTS)
  add_executable(cpp_smoke tests/test_smoke.cpp)
  add_test(NAME cpp_smoke COMMAND cpp_smoke)
endif()
